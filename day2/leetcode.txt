nums=list(map(int,input().split())
n=len(nums)
original_sum=n*(n+1)//2
arr_sum=sum(nums)
return original_sum-arr_sum

bool isPalindrome(int x) {
    long long rev=0;
    int duplicate=x;
    while(x>0)
    {
        int ld=x%10;
        x=x/10;
        rev=(rev*10)+ld;
    }
    if(rev==duplicate)
    {
        return true;
    }
    else
    {
        return false;
    }
}
