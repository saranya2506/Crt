class Solution:
    def numOfSubarrays(self, arr: list[int], k: int, threshold: int) -> int:
        i=0
        j=0
        sum=0
        c=0
        total=0
        while(j<len(arr)):
            sum+=arr[j]
            if(j-i+1<k):
                j+=1
            elif(j-i+1==k):
                total=sum/k
                if(total>=threshold):
                    c+=1
                sum-=arr[i]
                i+=1
                j+=1
        return c
